/* yw05-panes: where does split_cells' single-precision rounding differ
 * from double precision?  ratio = (float)want / (float)u1 (yew_pane_resize),
 * then cells = (i32)((float)u2 * ratio + 0.5f) (split_cells). */
#include <stdio.h>
int main(void) {
    int found = 0;
    for (int u1 = 25; u1 < 400 && found < 12; u1++)
        for (int want = 12; want <= u1 - 12 && found < 12; want++) {
            float rf = (float)want / (float)u1;
            double rd = (double)want / (double)u1;
            for (int u2 = 25; u2 < 2000; u2++) {
                int cf = (int)((float)u2 * rf + 0.5f);
                int cd = (int)((double)u2 * rd + 0.5);
                if (cf != cd) {
                    printf("u1=%d want=%d span=%d usable=%d float=%d double=%d\n", u1, want, u2 + 1, u2, cf, cd);
                    found++;
                    break;
                }
            }
        }
    {
        float rf = 62.0f / 63.0f;
        printf("check 62/63 at 1024: float=%d double=%d\n", (int)((float)1024 * rf + 0.5f), (int)(1024.0 * (62.0 / 63.0) + 0.5));
    }
    return 0;
}
