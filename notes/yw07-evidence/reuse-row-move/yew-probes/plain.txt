//! member: false
use edit
fn fill(mut menu: edit.Menu, n: int) {
    {
        var rows = List[edit.CompItem]()
        for i in 0..n {
            let bytes = "item-{i}".bytes()
            var row = edit.compitem_text(bytes, edit.COMP_CMD)
            row.detail = "detail-{i % 3}"
            row.is_dir = i % 2 == 0
            row.deferred = i % 3 == 0
            row.score = 100 - i
            row.match_off = 2
            (mut row.m.pos).push(0)
            (mut row.suffix).push(32 as byte)
            (mut rows).push(row)
        }
        edit.menu_refill(mut menu, rows)
    }
}

fn main() {
    var menu = edit.menu_new(edit.menuspec_default())
    print("before first")
    fill(mut menu, 2)
    print("after first: {menu.items.len}")
    fill(mut menu, 1)
    print("after shrink: {menu.items.len}")
    fill(mut menu, 2)
    print("after grow: {menu.items.len}")
}
